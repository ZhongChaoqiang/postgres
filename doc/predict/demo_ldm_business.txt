-- ============================================================
-- ldm_infer 业务演示：多列结构化数据智能分析
-- 使用 EMBEDDINGS AS 语法对多列结构化数据生成组合向量
-- 演示 ldm_infer 四种任务类型在真实业务场景中的应用
--
-- 与 EMBEDDING AS（单列文本嵌入）的区别：
--   EMBEDDING AS (st_embedding(col))        → 单列文本 → 向量
--   EMBEDDINGS AS (ft_transformer_embedding(col1, col2, ...)) → 多列组合 → 向量
-- ============================================================

-- ============================================================
-- 场景一：客户信用评级分类 (classification)
-- 业务场景：银行根据客户的职业、城市、年龄段、收入水平
--           自动评定信用等级（excellent/good/fair/poor）
-- 多列结构化数据：occupation + city + age_group + income_level
-- ============================================================

\echo '=== 场景一：客户信用评级分类 ==='
\echo '业务场景：银行客户信用自动评级（excellent/good/fair/poor）'
\echo '多列特征：occupation, city, age_group, income_level'

DROP TABLE IF EXISTS customer_credit CASCADE;
CREATE TABLE customer_credit (
    id serial PRIMARY KEY,
    occupation text,
    city text,
    age_group text,
    income_level text,
    credit_rating text PREDICT AS (ldm_infer()),
    customer_profile EMBEDDINGS AS (ft_transformer_embedding(occupation, city, age_group, income_level))
) WITH (predict_timing=immediate, ldm_task='classification', ldm_topn=3, vector_len=384);

-- 训练数据：历史客户及其信用评级（每类4条，特征语义相近）
INSERT INTO customer_credit (occupation, city, age_group, income_level, credit_rating) VALUES
    -- excellent: 高收入专业人士
    ('software architect', 'san francisco', 'middle', 'high', 'excellent'),
    ('investment banker', 'new york', 'middle', 'high', 'excellent'),
    ('surgeon', 'boston', 'senior', 'high', 'excellent'),
    ('tech company executive', 'seattle', 'senior', 'high', 'excellent'),
    -- good: 稳定中等收入
    ('software engineer', 'austin', 'young', 'medium', 'good'),
    ('registered nurse', 'denver', 'middle', 'medium', 'good'),
    ('accountant', 'portland', 'middle', 'medium', 'good'),
    ('high school teacher', 'chicago', 'middle', 'medium', 'good'),
    -- fair: 一般收入
    ('retail worker', 'phoenix', 'young', 'low', 'fair'),
    ('truck driver', 'dallas', 'middle', 'low', 'fair'),

    -- poor: 不稳定低收入
    ('unemployed', 'detroit', 'young', 'none', 'poor'),

-- 新客户：自动信用评级

    ('data scientist', 'san jose', 'young', 'high'),        -- 预期: excellent（高收入科技行业）

-- ============================================================
-- 场景二：房屋租金预测 (regression)
-- 业务场景：房产平台根据位置、房型、面积、装修情况
--           预测月租金（美元），用于租金定价建议
-- 多列结构化数据：location + house_type + size + furnishing
-- ============================================================

\echo ''
\echo '=== 场景二：房屋租金预测 ==='
\echo '业务场景：房产平台预测月租金（美元），用于定价建议'
\echo '多列特征：location, house_type, size, furnishing'

DROP TABLE IF EXISTS house_rent CASCADE;
CREATE TABLE house_rent (
    id serial PRIMARY KEY,
    location text,
    house_type text,
    size text,
    furnishing text,
    monthly_rent text PREDICT AS (ldm_infer()),
    property_features EMBEDDINGS AS (ft_transformer_embedding(location, house_type, size, furnishing))
) WITH (predict_timing=immediate, ldm_task='regression', ldm_topn=3, vector_len=384);

-- 训练数据：历史房源 → 实际月租金（美元）

    ('brooklyn new york', 'two bedroom apartment', 'large', 'furnished', '3000'),

-- ============================================================
-- 场景三：交易欺诈检测 (anomaly)
-- 业务场景：支付平台检测异常交易，识别潜在的欺诈行为
--           正常交易模式：常规商户、合理金额、常见支付方式
--           异常交易模式：非常规商户、异常金额、可疑支付方式
-- 多列结构化数据：merchant + category + amount + payment_method
-- ============================================================

\echo ''
\echo '=== 场景三：交易欺诈检测 ==='
\echo '业务场景：支付平台检测异常交易，识别潜在欺诈'
\echo '多列特征：merchant, category, amount, payment_method'

DROP TABLE IF EXISTS transaction_fraud CASCADE;
CREATE TABLE transaction_fraud (
    id serial PRIMARY KEY,
    merchant text,
    category text,
    amount text,
    payment_method text,
    status text PREDICT AS (ldm_infer()),
    transaction_features EMBEDDINGS AS (ft_transformer_embedding(merchant, category, amount, payment_method))
) WITH (predict_timing=immediate, ldm_task='anomaly', ldm_topn=3, vector_len=384);

-- ============================================================
-- 场景四：智能产品推荐 (extraction)
-- 业务场景：电商平台根据客户需求、预算、使用场景
--           匹配最相似的历史客户，复用其购买的产品推荐
-- 多列结构化数据：customer_need + budget_range + use_case
-- ============================================================

\echo ''
\echo '=== 场景四：智能产品推荐 ==='
\echo '业务场景：电商平台根据客户需求匹配最相似的历史推荐'
\echo '多列特征：customer_need, budget_range, use_case'

DROP TABLE IF EXISTS product_recommendation CASCADE;
CREATE TABLE product_recommendation (
    id serial PRIMARY KEY,
    customer_need text,
    budget_range text,
    use_case text,
    recommended_product text PREDICT AS (ldm_infer()),
    customer_profile EMBEDDINGS AS (ft_transformer_embedding(customer_need, budget_range, use_case))
) WITH (predict_timing=immediate, ldm_task='extraction', ldm_topn=1, vector_len=384);
